```typescript
import { FastifyRequest, FastifyReply } from 'fastify'
import prisma from '../../database'
import { AppError } from '../errors/AppError'

export async function authenticate(
  request: FastifyRequest,
  reply: FastifyReply
) {
  try {
    await request.jwtVerify()
  } catch {
    return reply.status(401).send({ message: 'Token inválido ou expirado' })
  }

  // Cross-tenant ownership check (D-08, D-09)
  // Only applies when the route has a :slug param.
  // Compare JWT.tenantId with the id of the tenant resolved from the URL slug.
  const { slug } = (request.params as Record<string, string | undefined>)
  if (slug) {
    const tenant = await prisma.tenant.findUnique({ where: { slug } })

    if (!tenant) {
      throw new AppError('Tenant não encontrado', 404)
    }

    if (request.user.tenantId !== tenant.id) {
      throw new AppError('Acesso negado a este tenant', 403)
    }
  }
}
```

Key points:
- `request.jwtVerify()` remains the first step — 401 on invalid/missing JWT (inline reply is correct here because jwtVerify() throws a Fastify-specific error, not an AppError)
- After verify, `request.user` is populated with the typed payload (sub, tenantId, role, name)
- `slug` is extracted from `request.params` cast to a partial record — safe even if the route has no :slug param (slug will be undefined, check is skipped)
- Prisma lookup resolves the slug to a Tenant record
- NOT FOUND and WRONG TENANT use `throw new AppError(...)` per D-05 — the global error handler registered in plan 01-01 serializes these correctly
- If JWT tenantId !== tenant.id → 403 (not 401 — token is valid, just wrong tenant)
- Routes WITHOUT a :slug param (e.g., POST /auth/login) are unaffected because `slug` is undefined
