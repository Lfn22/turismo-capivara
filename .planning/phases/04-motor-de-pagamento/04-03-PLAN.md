  ```typescript
  import { FastifyInstance } from 'fastify'
  import crypto from 'node:crypto'
  import { MercadoPagoConfig, Payment } from 'mercadopago'
  import prisma from '../../database'
  import { AppError } from '../../shared/errors/AppError'

  if (!process.env.MP_ACCESS_TOKEN) {
    throw new Error('MP_ACCESS_TOKEN environment variable is required')
  }

  if (!process.env.MP_WEBHOOK_SECRET) {
    throw new Error('MP_WEBHOOK_SECRET environment variable is required')
  }

  const client = new MercadoPagoConfig({
    accessToken: process.env.MP_ACCESS_TOKEN,
  })

  function validateMpSignature(
    rawBody: Buffer,
    signatureHeader: string,
    paymentId: string
  ): boolean {
    // Parse "ts=<timestamp>,v1=<hash>" from x-signature header
    const parts: Record<string, string> = {}
    for (const part of signatureHeader.split(',')) {
      const [key, value] = part.trim().split('=')
      if (key && value) parts[key] = value
    }

    const ts = parts['ts']
    const v1 = parts['v1']

    if (!ts || !v1) return false

    // Manifest: "id:<paymentId>;request-date:<ts>;"
    const manifest = `id:${paymentId};request-date:${ts};`

    const computed = crypto
      .createHmac('sha256', process.env.MP_WEBHOOK_SECRET!)
      .update(manifest)
      .digest('hex')

    try {
      return crypto.timingSafeEqual(
        Buffer.from(computed, 'hex'),
        Buffer.from(v1, 'hex')
      )
    } catch {
      // Buffers different lengths — invalid signature
      return false
    }
  }

  export async function webhooksRoutes(app: FastifyInstance) {
    app.post(
      '/webhooks/mercadopago',
      {
        config: { rawBody: true }, // Required for fastify-raw-body capture
      },
      async (request, reply) => {
        // 1. Validate x-signature header (per D-06)
        const signatureHeader = request.headers['x-signature']
        if (!signatureHeader || typeof signatureHeader !== 'string') {
          return reply.status(400).send({ message: 'Assinatura ausente' })
        }

        // 2. Parse body for data.id
        const body = request.body as { action?: string; data?: { id?: string } }
        const paymentId = body?.data?.id

        if (!paymentId) {
          return reply.status(400).send({ message: 'ID de pagamento ausente' })
        }

        // 3. Validate HMAC signature
        const rawBody = (request as any).rawBody as Buffer
        const isValid = validateMpSignature(rawBody, signatureHeader, paymentId)

        if (!isValid) {
          return reply.status(400).send({ message: 'Assinatura inválida' })
        }

        // 4. Fetch payment details from MP to get status and external_reference
        let payment
        try {
          payment = await new Payment(client).get({ id: paymentId })
        } catch {
          // MP unreachable — return 200 so MP doesn't retry flood
          app.log.error(`Falha ao buscar pagamento ${paymentId} no Mercado Pago`)
          return reply.status(200).send({ message: 'ok' })
        }

        const externalReference = payment.external_reference
        const status = payment.status

        if (!externalReference) {
          return reply.status(200).send({ message: 'ok' })
        }

        // 5. Find booking by external_reference (= booking.id set at creation)
        const booking = await prisma.booking.findFirst({
          where: { id: externalReference },
          include: { slot: true },
        })

        if (!booking) {
          // Unknown booking — return 200 (MP will retry 404 indefinitely)
          return reply.status(200).send({ message: 'ok' })
        }

        // 6. Idempotency check (per D-08)
        if (booking.status === 'CONFIRMED' || booking.status === 'EXPIRED') {
          return reply.status(200).send({ message: 'ok' })
        }

        // 7. Transition based on payment status
        if (status === 'approved') {
          await prisma.booking.update({
            where: { id: booking.id },
            data: { status: 'CONFIRMED' },
          })
        } else if (status === 'cancelled' || status === 'rejected') {
          // payment.cancelled or payment.expired — transition to EXPIRED and release slot (per D-04)
          await prisma.$transaction(async (tx) => {
            await tx.booking.update({
              where: { id: booking.id },
              data: { status: 'EXPIRED' },
            })

            await tx.departureSlot.update({
              where: { id: booking.slotId },
              data: {
                booked: { decrement: booking.pax },
                status: 'OPEN',
              },
            })
          })
        }
        // Other statuses (pending, in_process) — no-op, return 200

        return reply.status(200).send({ message: 'ok' })
      }
    )
  }
  ```

  Key implementation notes:
  - `config: { rawBody: true }` is required on this route because fastify-raw-body is registered with `global: false` (plan 04-01). Without this config, `request.rawBody` will be undefined.
  - HMAC manifest format: `id:<paymentId>;request-date:<ts>;` — exact MP 2024+ format (per key finding #10).
  - `crypto.timingSafeEqual` prevents timing attacks. Buffers must be same length — wrap in try/catch.
  - `external_reference` lookup avoids race condition where webhook fires before paymentId stored in booking (per key finding #3).
  - On MP fetch failure, return 200 to prevent MP retry flood. Log the error.
  - EXPIRED transition releases slot inside `$transaction` (per D-04).
  - `action` field from body is NOT used for routing — payment status from MP API is authoritative.

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Internet → POST /webhooks/mercadopago | Unauthenticated external requests; anyone can POST |
| x-signature header | Attacker-controlled string parsed for HMAC validation |
| MP API → booking lookup | External service response trusted for status transitions |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-03-01 | Spoofing | POST /webhooks/mercadopago (no JWT) | mitigate | HMAC-SHA256 with MP_WEBHOOK_SECRET validates every request before business logic; reject 400 if missing or invalid |
| T-04-03-02 | Tampering | x-signature header parsing | mitigate | Parse ts= and v1= strictly; validate both present before computing HMAC; use timingSafeEqual to prevent timing oracle |
| T-04-03-03 | Repudiation | Booking state transitions | mitigate | All transitions logged via Fastify logger; paymentId stored on booking for audit trail |
| T-04-03-04 | Information Disclosure | MP_WEBHOOK_SECRET in env | mitigate | Read from process.env only; throw startup error if missing; never logged or returned in responses |
| T-04-03-05 | Denial of Service | Webhook replay attack | mitigate | Idempotency check (D-08): already CONFIRMED/EXPIRED returns 200 without DB writes |
| T-04-03-06 | Elevation of Privilege | Booking status manipulation via spoofed webhook | mitigate | HMAC validation must pass before any status read or write; 400 on failure — no partial processing |
| T-04-03-07 | Tampering | rawBody Buffer vs parsed body | mitigate | HMAC computed over raw body; parsed body only used to extract paymentId for manifest; signature covers paymentId |
</threat_model>
