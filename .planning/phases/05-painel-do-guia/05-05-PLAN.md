```typescript
"use client"
import { useState, useEffect } from "react"
import { useSession } from "next-auth/react"
import Calendar from "react-calendar"
import { Modal } from "@/components/ui/Modal"

interface Slot {
  id: string
  packageId: string
  startsAt: string
  capacity: number
  booked: number
  status: "OPEN" | "CLOSED" | "DEPARTED"
}

interface Package {
  id: string
  name: string
}

const API_URL = process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:3333"

function toLocalDateStr(date: Date): string {
  // Formato YYYY-MM-DD no timezone local do usuário
  const y = date.getFullYear()
  const m = String(date.getMonth() + 1).padStart(2, "0")
  const d = String(date.getDate()).padStart(2, "0")
  return `${y}-${m}-${d}`
}

function formatTime(iso: string) {
  return new Date(iso).toLocaleTimeString("pt-BR", { hour: "2-digit", minute: "2-digit" })
}

export default function DisponibilidadePage({ params }: { params: { slug: string } }) {
  const { data: session } = useSession()
  const token = (session?.user as any)?.token ?? ""
  const slug = params.slug

  const [selectedDate, setSelectedDate] = useState<Date | null>(null)
  const [slots, setSlots] = useState<Slot[]>([])
  const [packages, setPackages] = useState<Package[]>([])
  const [loading, setLoading] = useState(false)
  const [closingSlot, setClosingSlot] = useState<string | null>(null)
  const [modalOpen, setModalOpen] = useState(false)

  // Modal form state
  const [formPackageId, setFormPackageId] = useState("")
  const [formStartTime, setFormStartTime] = useState("")
  const [formEndTime, setFormEndTime] = useState("")
  const [formVagas, setFormVagas] = useState<number>(1)
  const [formErrors, setFormErrors] = useState<Record<string, string>>({})
  const [formSubmitting, setFormSubmitting] = useState(false)
  const [formApiError, setFormApiError] = useState<string | null>(null)

  // Load packages for modal dropdown
  useEffect(() => {
    if (!token) return
    fetch(`${API_URL}/tenants/${slug}/packages`, {
      headers: { Authorization: `Bearer ${token}` },
    })
      .then((r) => r.json())
      .then((data) => {
        const pkgs = Array.isArray(data) ? data : data.packages ?? []
        setPackages(pkgs)
        if (pkgs.length > 0) setFormPackageId(pkgs[0].id)
      })
      .catch(() => {})
  }, [token, slug])

  // Load slots for selected date
  useEffect(() => {
    if (!selectedDate || !token || packages.length === 0) return
    setLoading(true)

    const dateStr = toLocalDateStr(selectedDate)

    // Buscar slots de todos os packages do guia e filtrar por data
    Promise.all(
      packages.map((pkg) =>
        fetch(`${API_URL}/tenants/${slug}/packages/${pkg.id}/slots`, {
          headers: { Authorization: `Bearer ${token}` },
        })
          .then((r) => r.json())
          .then((data) => (Array.isArray(data) ? data : data.slots ?? []) as Slot[])
          .catch(() => [] as Slot[])
      )
    )
      .then((allSlots) => {
        const flat = allSlots.flat()
        const forDate = flat.filter((s) => s.startsAt.startsWith(dateStr))
        setSlots(forDate)
      })
      .finally(() => setLoading(false))
  }, [selectedDate, token, packages, slug])

  // Compute dates with slots for calendar tiling
  const slotsWithDates = slots // simplified — for tile coloring, load all slots
  function getTileStyle(date: Date): React.CSSProperties {
    const dateStr = toLocalDateStr(date)
    const daySlots = slots.filter((s) => s.startsAt.startsWith(dateStr))
    if (daySlots.length === 0) return {}
    const fullyBooked = daySlots.every((s) => s.booked >= s.capacity || s.status !== "OPEN")
    if (fullyBooked) return { background: "rgba(220,38,38,0.08)", borderRadius: "4px" }
    return { background: "rgba(196,133,42,0.15)", borderRadius: "4px" }
  }

  async function handleCloseSlot(slot: Slot) {
    setClosingSlot(slot.id)
    try {
      const res = await fetch(
        `${API_URL}/tenants/${slug}/packages/${slot.packageId}/slots/${slot.id}/close`,
        { method: "PATCH", headers: { Authorization: `Bearer ${token}` } }
      )
      if (res.ok) {
        setSlots((prev) =>
          prev.map((s) => (s.id === slot.id ? { ...s, status: "CLOSED" } : s))
        )
      }
    } finally {
      setClosingSlot(null)
    }
  }

  function validateForm(): boolean {
    const errs: Record<string, string> = {}
    if (!formPackageId) errs.packageId = "Campo obrigatório."
    if (!formStartTime) errs.startTime = "Campo obrigatório."
    if (!formEndTime) errs.endTime = "Campo obrigatório."
    else if (formStartTime && formEndTime && formEndTime <= formStartTime) {
      errs.endTime = "O horário de fim deve ser após o início."
    }
    if (!formVagas || formVagas < 1) errs.vagas = "Informe ao menos 1 vaga."
    setFormErrors(errs)
    return Object.keys(errs).length === 0
  }

  async function handleCreateSlot(e: React.FormEvent) {
    e.preventDefault()
    if (!validateForm() || !selectedDate) return
    setFormSubmitting(true)
    setFormApiError(null)

    const dateStr = toLocalDateStr(selectedDate)
    const startsAt = new Date(`${dateStr}T${formStartTime}:00`).toISOString()

    try {
      const res = await fetch(
        `${API_URL}/tenants/${slug}/packages/${formPackageId}/slots`,
        {
          method: "POST",
          headers: {
            Authorization: `Bearer ${token}`,
            "Content-Type": "application/json",
          },
          body: JSON.stringify({ startsAt, capacity: formVagas }),
        }
      )
      if (!res.ok) {
        const err = await res.json().catch(() => ({ message: "Erro ao criar slot" }))
        throw new Error(err.message)
      }
      const data = await res.json()
      const newSlot: Slot = data.slot ?? data
      setSlots((prev) => [...prev, newSlot])
      setModalOpen(false)
      setFormStartTime("")
      setFormEndTime("")
      setFormVagas(1)
      setFormErrors({})
    } catch (err: any) {
      setFormApiError(err.message ?? "Não foi possível criar o slot. Tente novamente.")
    } finally {
      setFormSubmitting(false)
    }
  }

  const inputStyle: React.CSSProperties = {
    width: "100%",
    padding: "8px 12px",
    border: "1px solid var(--stone-300)",
    borderRadius: "4px",
    fontSize: "16px",
    color: "var(--stone-800)",
    background: "white",
  }

  const labelStyle: React.CSSProperties = {
    display: "block",
    fontSize: "14px",
    color: "var(--stone-700)",
    marginBottom: "4px",
    fontWeight: 600,
  }

  const errorStyle: React.CSSProperties = {
    fontSize: "14px",
    color: "#DC2626",
    marginTop: "4px",
  }

  return (
    <>
      {/* Page header */}
      <div style={{ marginBottom: "24px" }}>
        <p style={{ fontSize: "11px", color: "var(--ochre)", letterSpacing: "0.2em", textTransform: "uppercase", marginBottom: "8px" }}>
          Painel do Guia
        </p>
        <h1 style={{ fontFamily: "var(--font-display)", fontSize: "24px", color: "var(--stone-900)", lineHeight: 1.2 }}>
          Disponibilidade
        </h1>
      </div>

      {/* Two-column layout: calendar + slots panel */}
      <div
        style={{
          display: "grid",
          gridTemplateColumns: "3fr 2fr",
          gap: "24px",
          alignItems: "start",
        }}
      >
        {/* Calendar */}
        <div
          style={{
            background: "white",
            border: "1px solid var(--stone-200)",
            borderRadius: "8px",
            padding: "24px",
          }}
        >
          <Calendar
            value={selectedDate}
            onChange={(date) => {
              setSelectedDate(date as Date)
            }}
            tileStyle={({ date }) => getTileStyle(date)}
            locale="pt-BR"
          />
        </div>

        {/* Slots panel */}
        <div
          style={{
            background: "white",
            border: "1px solid var(--stone-200)",
            borderRadius: "8px",
            padding: "24px",
            minHeight: "200px",
          }}
        >
          {!selectedDate ? (
            <p style={{ fontSize: "16px", color: "var(--stone-500)", textAlign: "center", padding: "32px 0" }}>
              Selecione um dia no calendário para ver os slots.
            </p>
          ) : loading ? (
            <p style={{ fontSize: "14px", color: "var(--stone-400)", textAlign: "center", padding: "16px 0" }}>
              Carregando slots...
            </p>
          ) : (
```
