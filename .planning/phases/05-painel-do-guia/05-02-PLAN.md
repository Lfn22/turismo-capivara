**Criar apps/web/lib/auth.ts:**
```typescript
import { NextAuthOptions } from "next-auth"
import CredentialsProvider from "next-auth/providers/credentials"

export const authOptions: NextAuthOptions = {
  providers: [
    CredentialsProvider({
      name: "credentials",
      credentials: {
        email: { label: "Email", type: "email" },
        password: { label: "Senha", type: "password" },
        tenantSlug: { label: "Tenant", type: "text" },
      },
      async authorize(credentials) {
        if (!credentials?.email || !credentials?.password || !credentials?.tenantSlug) {
          return null
        }
        const res = await fetch(`${process.env.API_URL}/auth/login`, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            email: credentials.email,
            password: credentials.password,
            tenantSlug: credentials.tenantSlug,
          }),
        })
        if (!res.ok) return null
        const { token } = await res.json()
        const payload = JSON.parse(
          Buffer.from(token.split(".")[1], "base64url").toString()
        )
        return {
          id: payload.sub,
          name: payload.name ?? credentials.email,
          email: credentials.email,
          role: payload.role,
          tenantId: payload.tenantId,
          token,
        }
      },
    }),
  ],
  callbacks: {
    async jwt({ token, user }) {
      if (user) {
        token.apiToken = (user as any).token
        token.role = (user as any).role
        token.tenantId = (user as any).tenantId
      }
      return token
    },
    async session({ session, token }) {
      session.user = {
        ...session.user,
        id: token.sub as string,
        role: token.role as string,
        tenantId: token.tenantId as string,
        token: token.apiToken as string,
      } as any
      return session
    },
  },
  pages: {
    signIn: "/login",
  },
  session: { strategy: "jwt" },
  secret: process.env.NEXTAUTH_SECRET,
}
```

**Criar apps/web/middleware.ts** (na raiz de apps/web, não dentro de app/):
```typescript
import { withAuth } from "next-auth/middleware"
import { NextResponse } from "next/server"

export default withAuth(
  function middleware(req) {
    const token = req.nextauth.token
    const pathname = req.nextUrl.pathname

    // Extrair slug do path (ex: /serra-viva/painel/dashboard)
    const slugMatch = pathname.match(/^\/([^/]+)\//)
    const slug = slugMatch?.[1] ?? ""

    if (pathname.includes("/painel") && token?.role !== "CONDUTOR") {
      return NextResponse.redirect(
        new URL(`/${slug}/login?error=forbidden`, req.url)
      )
    }

    if (pathname.includes("/admin") && token?.role !== "ADMIN") {
      return NextResponse.redirect(
        new URL(`/${slug}/login?error=forbidden`, req.url)
      )
    }

    return NextResponse.next()
  },
  {
    callbacks: {
      authorized: ({ token }) => !!token,
    },
  }
)

export const config = {
  matcher: ["/:slug/painel/:path*", "/:slug/admin/:path*"],
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → NextAuth route | /api/auth/* recebe credenciais — HTTPS em produção |
| NextAuth → Fastify API | authorize() faz fetch server-side com credenciais — não expõe ao browser |
| Cookie NextAuth → Client | httpOnly cookie — não acessível por JavaScript |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-05 | Spoofing | login page → signIn() | mitigate | next-auth CSRF token embutido em todas as calls de signIn |
| T-05-06 | Elevation of Privilege | middleware role check | mitigate | `token?.role !== "CONDUTOR"` verifica role do JWT; role vem do JWT assinado pela API, não de input do cliente |
| T-05-07 | Information Disclosure | NEXTAUTH_SECRET | mitigate | Variável server-only; nunca exposta como NEXT_PUBLIC_; verificar .gitignore |
| T-05-08 | Tampering | JWT session cookie | accept | next-auth usa NEXTAUTH_SECRET para assinar o cookie JWT — adulteração invalida a assinatura |
</threat_model>
