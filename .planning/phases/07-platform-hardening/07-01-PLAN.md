    Build a dedicated test app (NOT using buildApp() helper) that registers

    ```typescript
    import { describe, it, expect, beforeAll, afterAll } from 'vitest'
    import Fastify, { FastifyInstance } from 'fastify'
    import rateLimit from '@fastify/rate-limit'
    import { authRoutes } from '../modules/auth/auth.routes'
    import { bookingsRoutes } from '../modules/bookings/bookings.routes'
    import { webhooksRoutes } from '../modules/webhooks/webhooks.routes'
    import { AppError } from '../shared/errors/AppError'

    // Build a test app with rate-limit active.
    // Production thresholds used directly — tests fire max+1 requests.
    async function buildRateLimitApp(): Promise<FastifyInstance> {
      const app = Fastify({ logger: false, trustProxy: true })

      await app.register(rateLimit, {
        global: true,
        max: 20,
        timeWindow: '1 minute',
        addHeaders: {
          'retry-after': true,
        },
      })

      // Register routes (they apply per-route config overrides)
      await app.register(authRoutes)
      await app.register(bookingsRoutes)
      await app.register(webhooksRoutes)

      app.get('/health', async () => ({ status: 'ok' }))

      app.setErrorHandler((err, _req, reply) => {
        if (err instanceof AppError) {
          return reply.status(err.statusCode).send({ message: err.message })
        }
        reply.status(500).send({ message: 'Erro interno' })
      })

      await app.ready()
      return app
    }

    describe('Rate Limiting', () => {
      let app: FastifyInstance

      beforeAll(async () => {
        // Set required env vars for route registration
        process.env.JWT_SECRET = 'test-secret-minimum-32-characters-xx'
        process.env.CPF_SECRET = 'test-cpf-secret-32-chars-xxxxxxxxx'
        process.env.MP_ACCESS_TOKEN = 'TEST_ACCESS_TOKEN'
        process.env.MP_WEBHOOK_SECRET = 'test-webhook-secret'
        app = await buildRateLimitApp()
      })

      afterAll(async () => { await app.close() })

      it('blocks 21st request to auth/login (max 20/min per IP)', async () => {
        for (let i = 0; i < 20; i++) {
          await app.inject({ method: 'POST', url: '/test-tenant/auth/login', payload: {} })
        }
        const res = await app.inject({ method: 'POST', url: '/test-tenant/auth/login', payload: {} })
        expect(res.statusCode).toBe(429)
      })

      it('blocks 21st request to auth/register (max 20/min per IP)', async () => {
        // Use different IP via x-forwarded-for to avoid counter bleed from login test
        for (let i = 0; i < 20; i++) {
          await app.inject({
            method: 'POST', url: '/test-tenant/auth/register', payload: {},
            headers: { 'x-forwarded-for': '10.0.0.2' }
          })
        }
        const res = await app.inject({
          method: 'POST', url: '/test-tenant/auth/register', payload: {},
          headers: { 'x-forwarded-for': '10.0.0.2' }
        })
        expect(res.statusCode).toBe(429)
      })

      it('blocks 61st request to POST /tenants/:slug/bookings (max 60/min per IP)', async () => {
        for (let i = 0; i < 60; i++) {
          await app.inject({
            method: 'POST', url: '/tenants/test-tenant/bookings', payload: {},
            headers: { 'x-forwarded-for': '10.0.0.3' }
          })
        }
        const res = await app.inject({
          method: 'POST', url: '/tenants/test-tenant/bookings', payload: {},
          headers: { 'x-forwarded-for': '10.0.0.3' }
        })
        expect(res.statusCode).toBe(429)
      })

      it('webhook route never returns 429 regardless of volume', async () => {
        const responses: number[] = []
        for (let i = 0; i < 30; i++) {
          const res = await app.inject({
            method: 'POST', url: '/webhooks/mercadopago',
            payload: {},
            headers: { 'x-forwarded-for': '10.0.0.4' }
          })
          responses.push(res.statusCode)
        }
        expect(responses.every(s => s !== 429)).toBe(true)
      })

      it('429 response includes Retry-After header', async () => {
        // Exceed auth limit
        for (let i = 0; i < 20; i++) {
          await app.inject({
            method: 'POST', url: '/test-tenant/auth/login', payload: {},
            headers: { 'x-forwarded-for': '10.0.0.5' }
          })
        }
        const res = await app.inject({
          method: 'POST', url: '/test-tenant/auth/login', payload: {},
          headers: { 'x-forwarded-for': '10.0.0.5' }
        })
        expect(res.statusCode).toBe(429)
        expect(res.headers['retry-after']).toBeDefined()
      })

      it('GET /health never rate-limited (30 requests all succeed)', async () => {
        const responses: number[] = []
        for (let i = 0; i < 30; i++) {
          const res = await app.inject({
            method: 'GET', url: '/health',
            headers: { 'x-forwarded-for': '10.0.0.6' }
          })
          responses.push(res.statusCode)
        }
        expect(responses.every(s => s !== 429)).toBe(true)
      })
    })
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Internet → Fastify | Untrusted HTTP requests from any IP; Railway reverse proxy sits in front |
| Railway proxy → Fastify | X-Forwarded-For header carries real client IP; must be trusted |
| Mercado Pago → /webhooks | Trusted payment processor; must never be rate-limited |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-7-01 | Denial of Service / Brute Force | POST /auth/login, POST /auth/register | mitigate | @fastify/rate-limit max:20/min per IP; 21st request returns 429 with Retry-After header |
| T-7-03 | Spoofing (IP bypass) | @fastify/rate-limit IP extraction | mitigate | trustProxy:true on Fastify so X-Forwarded-For is trusted; Railway sets this header authoritatively |
| T-7-04 | Denial of Service (payment disruption) | POST /webhooks/mercadopago | mitigate | config:{ rateLimit: false } on webhook route — MP retries will never be blocked |
</threat_model>
