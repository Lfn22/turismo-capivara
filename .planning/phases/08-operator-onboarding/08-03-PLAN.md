const API_URL = process.env.API_URL ?? process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:3333"

export async function GET(req: NextRequest) {
  const jwt = await getToken({ req, secret: process.env.NEXTAUTH_SECRET })
  if (!jwt?.apiToken) return NextResponse.json({ message: "Não autenticado" }, { status: 401 })
  const res = await fetch(`${API_URL}/tenants/admin/pending`, {
    headers: { Authorization: `Bearer ${jwt.apiToken}` },
    cache: "no-store",
  })
  const data = await res.text()
  return new NextResponse(data, { status: res.status, headers: { "Content-Type": "application/json" } })
}

**Step A — Update `apps/web/middleware.ts`:**

Add SUPER_ADMIN protection inside the middleware function, after the existing `/admin` check and before `return NextResponse.next()`:

```typescript
if (pathname.startsWith('/super-admin') && token?.role !== 'SUPER_ADMIN') {
  return NextResponse.redirect(new URL('/login?error=forbidden', req.url))
}
```

Note: redirect target is `/login` (root-level, not slug-based). If no `/login` page exists, redirect to `/` instead — check before deciding.

Extend the config matcher to include `/super-admin/:path*`:
```typescript
export const config = {
  matcher: [
    "/:slug/painel/:path*",
    "/:slug/admin/:path*",
    "/super-admin/:path*",
  ],
}
```

CRITICAL: The slug extraction (`pathname.match(/^\/([^/]+)\//)`) for /super-admin paths will extract "super-admin" as the slug value. This is used in the `!token` redirect: `/${slug}/login` would produce `/super-admin/login` — an invalid page. Guard this: when `pathname.startsWith('/super-admin')` and `!token`, redirect to `/` or a dedicated auth page, NOT `/${slug}/login`. The cleanest fix is to add a guard before the slug-based redirect:

```typescript
if (!token) {
  if (pathname.startsWith('/super-admin')) {
    return NextResponse.redirect(new URL('/', req.url))
  }
  return NextResponse.redirect(
    new URL(`/${slug}/login?callbackUrl=${encodeURIComponent(req.url)}`, req.url)
  )
}

// slugify helper
function slugify(str: string): string {
  return str
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-|-$/g, '')
}

async function handleApprove(tenant: Tenant) {
  setActionLoading(tenant.id)
  setActionError(null)
  setItems((prev) => prev.map((t) => t.id === tenant.id ? { ...t, approvalStatus: "APPROVED" } : t))
  try {
    const res = await fetch(`/api/super-admin/tenants/${tenant.id}/approve`, { method: "PATCH" })
    if (!res.ok) throw new Error()
  } catch {
    setItems((prev) => prev.map((t) => t.id === tenant.id ? { ...t, approvalStatus: "PENDING" } : t))
    setActionError("Não foi possível realizar a ação. Tente novamente.")
  } finally {
    setActionLoading(null)
  }
}
```

**Reject handler:**
```typescript
async function handleRejectSubmit() {
  if (!rejectTarget || !rejectReason.trim()) {
    setRejectError("Informe o motivo da rejeição.")
    return
  }
  setRejectSubmitting(true)
  setRejectError(null)
  try {
    const res = await fetch(`/api/super-admin/tenants/${rejectTarget.id}/reject`, {
      method: "PATCH",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ reason: rejectReason }),
    })
    if (!res.ok) throw new Error()
    setItems((prev) => prev.map((t) => t.id === rejectTarget.id ? { ...t, approvalStatus: "REJECTED", rejectionReason: rejectReason } : t))
    setRejectTarget(null)
    setRejectReason("")
  } catch {
    setRejectError("Não foi possível rejeitar a operadora. Tente novamente.")
  } finally {
    setRejectSubmitting(false)
  }
}
```
