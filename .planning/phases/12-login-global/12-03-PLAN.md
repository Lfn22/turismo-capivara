Add three new async functions to auth-client.ts that call the backend routes.

```typescript
export async function lookupTenant(email: string): Promise<{ tenantName: string; tenantSlug: string } | null> {
  try {
    const res = await fetch('/api/auth/lookup-tenant', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email }),
    });

    if (!res.ok) {
      throw new Error(`lookup-tenant failed: ${res.status}`);
    }

    const data = await res.json();
    
    // Check if we got a real result or just a generic message
    if (!data.tenantSlug) {
      return null; // Unknown email (or generic response)
    }

    return {
      tenantName: data.tenantName,
      tenantSlug: data.tenantSlug,
    };
  } catch (error) {
    console.error('lookupTenant error:', error);
    throw error;
  }
}

export async function requestPasswordReset(email: string): Promise<void> {
  try {
    const res = await fetch('/api/auth/request-password-reset', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email }),
    });

    if (!res.ok) {
      throw new Error(`request-password-reset failed: ${res.status}`);
    }

    // Success (always returns 200, even for unknown emails per D-06)
  } catch (error) {
    console.error('requestPasswordReset error:', error);
    throw error;
  }
}

export async function resetPassword(userId: string, token: string, newPassword: string): Promise<void> {
  try {
    const res = await fetch('/api/auth/reset-password', {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        userId,
        token,
        newPassword,
      }),
    });

    if (!res.ok) {
      const data = await res.json();
      throw new Error(data.error || 'Reset password failed');
    }

    // Success
  } catch (error) {
    console.error('resetPassword error:', error);
    throw error;
  }
}
```

```typescript
'use client';

import { useState } from 'react';
import { useRouter } from 'next/navigation';
import Link from 'next/link';
import { lookupTenant, verifyCredentials } from '@/lib/auth-client';

export default function LoginPage() {
  const router = useRouter();
  const [step, setStep] = useState<'email' | 'password'>('email');
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [tenantName, setTenantName] = useState('');
  const [tenantSlug, setTenantSlug] = useState('');
  const [error, setError] = useState('');
  const [isLoading, setIsLoading] = useState(false);

  const handleEmailSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setError('');
    setIsLoading(true);

    try {
      // Step 1: Look up tenant by email (per D-03)
      const result = await lookupTenant(email);
      
      if (!result) {
        setError('Email não encontrado');
        setIsLoading(false);
        return;
      }

      // Email found: move to Step 2 with tenant resolved
      setTenantName(result.tenantName);
      setTenantSlug(result.tenantSlug);
      setStep('password');
    } catch (err) {
      setError('Erro ao buscar tenant. Tente novamente.');
      console.error(err);
    } finally {
      setIsLoading(false);
    }
  };

  const handlePasswordSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setError('');
    setIsLoading(true);

    try {
      // Step 2: Verify credentials with resolved tenantSlug (per D-04)
      const result = await verifyCredentials(email, password, tenantSlug);
      
      // Success: store token and redirect to dashboard
      localStorage.setItem('authToken', result.token);
      router.push(`/${tenantSlug}/dashboard`);
    } catch (err) {
      setError('Email ou senha incorretos');
      console.error(err);
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <main className="min-h-screen flex items-center justify-center bg-gradient-to-b from-blue-50 to-white p-4">
      <div className="card w-full max-w-md bg-base-100 shadow-xl">
        <div className="card-body">
          <h1 className="text-2xl font-bold mb-6 text-center">CAPI</h1>

          {step === 'email' ? (
            <>
              <h2 className="text-xl font-semibold mb-4">Bem-vindo</h2>
              <form onSubmit={handleEmailSubmit}>
                <div className="form-control mb-4">
                  <label className="label">
                    <span className="label-text">Email</span>
                  </label>
                  <input
                    type="email"
                    placeholder="seu@email.com"
                    className="input input-bordered"
                    value={email}
                    onChange={(e) => setEmail(e.target.value)}
                    disabled={isLoading}
                    required
                  />
                </div>
                {error && <div className="alert alert-error mb-4">{error}</div>}
                <button
                  type="submit"
                  className="btn btn-primary w-full"
                  disabled={isLoading}
                >
                  {isLoading ? 'Buscando...' : 'Continuar'}
                </button>
              </form>

              <div className="divider">ou</div>

              {/* Per LOGIN-04: Link to onboarding */}
              <p className="text-sm text-center">
                Cadastrar agência ou guia?{' '}
                <Link href="/onboarding" className="link link-primary">
                  Comece aqui
                </Link>
              </p>

              <p className="text-sm text-center mt-4">
                <Link href="/forgot-password" className="link link-primary">
                  Esqueceu a senha?
                </Link>
              </p>
            </>
          ) : (
            <>
              <h2 className="text-xl font-semibold mb-2">Bem-vindo, {tenantName}</h2>
              <p className="text-sm text-gray-500 mb-4">{email}</p>
              <form onSubmit={handlePasswordSubmit}>
                <div className="form-control mb-4">
                  <label className="label">
                    <span className="label-text">Senha</span>
                  </label>
                  <input
                    type="password"
                    placeholder="••••••••"
                    className="input input-bordered"
                    value={password}
                    onChange={(e) => setPassword(e.target.value)}
                    disabled={isLoading}
                    required
                  />
                </div>
                {error && <div className="alert alert-error mb-4">{error}</div>}
                <button
                  type="submit"
                  className="btn btn-primary w-full"
                  disabled={isLoading}
                >
                  {isLoading ? 'Entrando...' : 'Entrar'}
                </button>
              </form>

              {/* Per D-02: "Não é você?" link to go back to step 1 */}
              <button
                type="button"

                onClick={() => {
```

Per D-08, D-09: Create /reset-password page that accepts userId and token in query params.

```typescript
'use client';

import { useState, useEffect } from 'react';
import { useRouter, useSearchParams } from 'next/navigation';
import Link from 'next/link';
import { resetPassword } from '@/lib/auth-client';

export default function ResetPasswordPage() {
  const router = useRouter();
  const searchParams = useSearchParams();
  const userId = searchParams.get('userId');
  const token = searchParams.get('token');

  const [newPassword, setNewPassword] = useState('');
  const [confirmPassword, setConfirmPassword] = useState('');
  const [error, setError] = useState('');
  const [success, setSuccess] = useState(false);
  const [isLoading, setIsLoading] = useState(false);

  useEffect(() => {
    if (!userId || !token) {
      setError('Link de redefinição inválido');
    }
  }, [userId, token]);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setError('');

    if (newPassword !== confirmPassword) {
      setError('Senhas não correspondem');
      return;
    }

    if (newPassword.length < 8) {
      setError('Senha deve ter no mínimo 8 caracteres');
      return;
    }

    setIsLoading(true);

    try {
      if (!userId || !token) {
        setError('Link inválido');
        return;
      }

      // Call resetPassword API (per D-09: after reset, redirect to /login)
      await resetPassword(userId, token, newPassword);
      
      setSuccess(true);
      // Redirect to login after success message (per D-09)
      setTimeout(() => {
        router.push('/login');
      }, 2000);
    } catch (err) {
      setError((err as Error).message || 'Erro ao redefinir senha');
      console.error(err);
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <main className="min-h-screen flex items-center justify-center bg-gradient-to-b from-blue-50 to-white p-4">
      <div className="card w-full max-w-md bg-base-100 shadow-xl">
        <div className="card-body">
          <h1 className="text-2xl font-bold mb-6 text-center">CAPI</h1>
          <h2 className="text-xl font-semibold mb-4">Redefinir Senha</h2>

          {!userId || !token ? (
            <>
              <div className="alert alert-error">Link de redefinição inválido ou expirado</div>
              <p className="text-sm text-center mt-4">
                <Link href="/login" className="link link-primary">
                  Voltar para login
                </Link>
              </p>
            </>
          ) : success ? (
            <>
              <div className="alert alert-success">Senha redefinida com sucesso!</div>
              <p className="text-sm text-center mt-4">Redirecionando para login...</p>
            </>
          ) : (
            <form onSubmit={handleSubmit}>
              <div className="form-control mb-4">
                <label className="label">
                  <span className="label-text">Nova Senha</span>
                </label>
                <input
                  type="password"
                  placeholder="••••••••"
                  className="input input-bordered"
                  value={newPassword}
                  onChange={(e) => setNewPassword(e.target.value)}
                  disabled={isLoading}
                  required
                />
                <label className="label">
                  <span className="label-text-alt">Mínimo 8 caracteres</span>
                </label>
              </div>

              <div className="form-control mb-4">
                <label className="label">
                  <span className="label-text">Confirmar Senha</span>
                </label>
                <input
                  type="password"
                  placeholder="••••••••"
                  className="input input-bordered"
                  value={confirmPassword}
                  onChange={(e) => setConfirmPassword(e.target.value)}
                  disabled={isLoading}
                  required
                />
              </div>

              {error && <div className="alert alert-error mb-4">{error}</div>}

              <button
                type="submit"
                className="btn btn-primary w-full"
                disabled={isLoading}
              >
                {isLoading ? 'Redefinindo...' : 'Redefinir Senha'}
              </button>
            </form>
          )}
        </div>
      </div>
    </main>
  );
}
```

| T-12-15 | Tampering | Query params (userId, token) | mitigate | Both validated on backend before use; missing/invalid params return error |
| T-12-16 | Repudiation | Login/reset flow | accept | User initiated; no session hijacking vectors in this phase |
