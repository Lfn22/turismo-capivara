```typescript
async hashToken(token: string): Promise<string> {
  return bcryptjs.hash(token, 10);
}

async verifyToken(userId: string, token: string, hashedToken: string): Promise<boolean> {
  // Do NOT use bcrypt.compare for token lookup (full-table scan pitfall per RESEARCH.md)
  // Instead: caller provides userId + token, we verify hash matches
  return bcryptjs.compare(token, hashedToken);
}

async lookupTenant(email: string): Promise<{ tenantName: string; tenantSlug: string } | null> {
  // Per D-06: Return generic message for unknown emails (but don't expose in service)
  // Service returns null; route handles generic response
  const user = await this.prisma.user.findUnique({
    where: { email },
    select: { id: true, tenant: { select: { name: true, slug: true } } },
  });
  
  if (!user) return null;
  
  return {
    tenantName: user.tenant.name,
    tenantSlug: user.tenant.slug,
  };
}

async requestPasswordReset(email: string): Promise<void> {
  const user = await this.prisma.user.findUnique({
    where: { email },
    select: { id: true, email: true, tenant: { select: { slug: true } } },
  });
  
  if (!user) {
    // Per D-06: Don't expose that email doesn't exist. Fire-and-forget.
    return;
  }
  
  // Generate raw token
  const rawToken = crypto.randomBytes(32).toString('hex');
  const hashedToken = await this.hashToken(rawToken);
  
  // Store token in DB (expires in 1 hour per D-08)
  await this.prisma.passwordResetToken.create({
    data: {
      userId: user.id,
      hashedToken,
      expiresAt: new Date(Date.now() + 60 * 60 * 1000), // 1 hour
    },
  });
  
  // Send email with reset link (fire-and-forget per PATTERNS.md)
  const resetUrl = \`\${process.env.NEXT_PUBLIC_APP_URL}/reset-password?userId=\${user.id}&token=\${rawToken}\`;
  void getResend().emails.send({
    from: 'CAPI <noreply@capi.local>',
    to: user.email,
    subject: 'Redefinir sua senha — CAPI',
    html: \`<p>Clique <a href="\${resetUrl}">aqui para redefinir sua senha</a>.</p><p>Este link expira em 1 hora.</p>\`,
  });
}

async resetPassword(userId: string, token: string, newPassword: string): Promise<boolean> {
  // Find token by userId (avoid bcrypt full-table scan per RESEARCH.md pitfall 1)
  const storedToken = await this.prisma.passwordResetToken.findFirst({
    where: {
      userId,
      expiresAt: { gt: new Date() }, // Not expired
      usedAt: null, // Not already used
    },
  });
  
  if (!storedToken) return false;
  
  // Verify hash
  const isValid = await this.verifyToken(userId, token, storedToken.hashedToken);
  if (!isValid) return false;
  
  // Hash new password
  const passwordHash = await bcryptjs.hash(newPassword, 10);
  
  // Update password and mark token as used (atomic)
  await this.prisma.\$transaction([
    this.prisma.user.update({
      where: { id: userId },
      data: { password: passwordHash },
    }),
    this.prisma.passwordResetToken.update({
      where: { id: storedToken.id },
      data: { usedAt: new Date() },
    }),
  ]);
  
  return true;
}
```

```typescript
import { FastifyRequest, FastifyReply } from 'fastify';
import { LookupTenantSchema } from '../schemas';
import { AuthService } from '../auth.service';

export default async function lookupTenant(
  request: FastifyRequest,
  reply: FastifyReply,
  authService: AuthService,
) {
  try {
    const body = LookupTenantSchema.parse(request.body);
    
    const result = await authService.lookupTenant(body.email);
    
    // Per D-06: Return generic message for unknown emails (anti-enumeration)
    if (!result) {
      return reply.code(200).send({
        message: 'Se o email estiver registrado, você receberá instruções em breve',
      });
    }
    
    // Email found: return tenant info
    return reply.code(200).send({
      tenantName: result.tenantName,
      tenantSlug: result.tenantSlug,
      email: body.email,
    });
  } catch (error) {
    if (error instanceof z.ZodError) {
      return reply.code(400).send({ error: 'Email inválido' });
    }
    request.log.error(error, 'lookup-tenant error');
    return reply.code(500).send({ error: 'Erro ao buscar tenant' });
  }
}
```

```typescript
import { FastifyRequest, FastifyReply } from 'fastify';
import { ResetPasswordSchema } from '../schemas';
import { AuthService } from '../auth.service';
import { z } from 'zod';

export default async function resetPassword(
  request: FastifyRequest,
  reply: FastifyReply,
  authService: AuthService,
) {
  try {
    const body = ResetPasswordSchema.parse(request.body);
    
    const success = await authService.resetPassword(
      body.userId,
      body.token,
      body.newPassword,
    );
    
    if (!success) {
      return reply.code(400).send({
        error: 'Link de redefinição inválido ou expirado',
      });
    }
    
    return reply.code(200).send({
      message: 'Senha redefinida com sucesso. Faça login com sua nova senha.',
      redirectUrl: '/login', // Per D-09: no auto-login, redirect to login
    });
  } catch (error) {
    if (error instanceof z.ZodError) {
      return reply.code(400).send({ error: 'Dados inválidos' });
    }
    request.log.error(error, 'reset-password error');
    return reply.code(500).send({ error: 'Erro ao redefinir senha' });
  }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Email Input → API | Untrusted user-supplied email addresses |
| Token → Reset API | Raw token in request body (sent from client, originally via email) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-12-06 | Tampering | Request body email/token | mitigate | Zod schema validation on all inputs; trim and validate email format |
| T-12-07 | Information Disclosure | Enum email addresses | mitigate | lookup-tenant and request-password-reset return generic messages for unknown emails (D-06) |
| T-12-08 | Tampering | Password reset token | mitigate | Token hashed with bcryptjs; lookup by userId + hash comparison (not full-table scan per pitfall 1) |
| T-12-09 | Denial of Service | Email spam via reset requests | mitigate | Rate limit: 3 requests per hour per IP for request-password-reset |
| T-12-10 | Tampering | Token reuse | mitigate | usedAt field set on consumption; future requests fail token lookup |
| T-12-11 | Repudiation | Password reset audit | accept | Password resets logged via request.log.error on errors; success not logged to avoid info disclosure |

</threat_model>
