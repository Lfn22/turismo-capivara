```typescript
import { NextRequest, NextResponse } from 'next/server'
import { getToken } from 'next-auth/jwt'

export async function GET(
  request: NextRequest,
  { params }: { params: Promise<{ slug: string; id: string }> }
) {
  const { slug, id } = await params
  const email = request.nextUrl.searchParams.get('email')

  if (!email) {
    return NextResponse.json({ error: 'email obrigatório' }, { status: 400 })
  }

  const apiBase = process.env.API_URL ?? 'http://localhost:3333'
  const url = `${apiBase}/tenants/${slug}/bookings/${id}?email=${encodeURIComponent(email)}`

  const res = await fetch(url, { cache: 'no-store' })
  if (!res.ok) {
    return NextResponse.json({ error: 'Reserva não encontrada' }, { status: res.status })
  }

  const data = await res.json()
  // Retorna apenas o status — evita expor dados sensíveis no polling
  return NextResponse.json({ status: data.status })
}
```

```typescript
function isValidCPF(cpf: string): boolean {
  if (/^(\d)\1{10}$/.test(cpf)) return false
  const digits = cpf.split('').map(Number)
  let sum = 0
  for (let i = 0; i < 9; i++) sum += digits[i] * (10 - i)
  let remainder = sum % 11
  const digit1 = remainder < 2 ? 0 : 11 - remainder
  if (digits[9] !== digit1) return false
  sum = 0
  for (let i = 0; i < 10; i++) sum += digits[i] * (11 - i)
  remainder = sum % 11
  const digit2 = remainder < 2 ? 0 : 11 - remainder
  return digits[10] === digit2
}
```
