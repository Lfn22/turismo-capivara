```typescript
// HARDENING-01: Idempotency check — deve vir ANTES do $transaction
const idempotencyKey = (request.headers['idempotency-key'] as string | undefined)?.trim() || undefined

if (idempotencyKey) {
  const existing = await prisma.booking.findUnique({
    where: { idempotencyKey },
  })
  if (existing && existing.tenantId === tenant.id) {
    return reply.status(200).send({
      id: existing.id,
      tenantId: existing.tenantId,
      slotId: existing.slotId,
      customerName: existing.customerName,
      customerEmail: existing.customerEmail,
      pax: existing.pax,
      status: existing.status,
      paymentId: existing.paymentId,
      paymentUrl: existing.paymentUrl,
      expiresAt: existing.expiresAt,
      createdAt: existing.createdAt,
      qrCode: existing.qrCode,
    })
  }
}
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-12.1-01 | Spoofing | Header Idempotency-Key | mitigate | Lookup faz `existing.tenantId === tenant.id` — chave de outro tenant não retorna booking errado |
| T-12.1-02 | Denial of Service | prisma.booking.findUnique por idempotencyKey | accept | Index @unique implícito no PostgreSQL — lookup é O(1), sem custo relevante |
| T-12.1-03 | Information Disclosure | Resposta de dedup retorna dados do booking | accept | Retorna apenas campos que o criador já recebeu no 201 original; sem dados sensíveis adicionais |
</threat_model>
