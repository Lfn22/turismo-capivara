```typescript
import { NextRequest, NextResponse } from "next/server"

const MAPTILER_KEY = process.env.MAPTILER_KEY

export async function GET(
  req: NextRequest,
  { params }: { params: Promise<{ path: string[] }> }
) {
  if (!MAPTILER_KEY) {
    return NextResponse.json({ message: "Configuração ausente" }, { status: 503 })
  }

  const { path } = await params
  const tilePath = path.join("/")

  // Bloquear path traversal
  if (tilePath.includes("..") || tilePath.includes("//")) {
    return NextResponse.json({ message: "Path inválido" }, { status: 400 })
  }

  // Preservar query string original (exceto key)
  const { searchParams } = new URL(req.url)
  searchParams.delete("key") // garantir que key do cliente não vaza
  const qs = searchParams.toString()

  const upstream = `https://api.maptiler.com/${tilePath}?key=${MAPTILER_KEY}${qs ? "&" + qs : ""}`

  let res: Response
  try {
    res = await fetch(upstream, {
      cache: "force-cache", // tiles são imutáveis por URL
      next: { revalidate: 86400 }, // 24h cache
    })
  } catch {
    return NextResponse.json({ message: "Serviço indisponível" }, { status: 502 })
  }

  const body = await res.arrayBuffer() // suporta tiles binários (PNG, PBF)
  return new NextResponse(body, {
    status: res.status,
    headers: {
      "Content-Type": res.headers.get("Content-Type") ?? "application/octet-stream",
      "Cache-Control": "public, max-age=86400, immutable",
    },
  })
}
```

IMPORTANTE:
- Usar `arrayBuffer()` (não `text()`) — tiles podem ser binários (PNG, PBF vector tiles)
- Nunca logar ou retornar `MAPTILER_KEY` em mensagens de erro
- Apenas método GET exportado (tiles são read-only)
- Cache 24h nos tiles para não desperdiçar quota free tier Maptiler

```typescript
'use client'

import { useRef, useEffect } from 'react'

export interface PartnerData {
  id: string
  name: string
  lat: number
  lng: number
}

interface MapWidgetProps {
  lat: number
  lng: number
  partners: PartnerData[]
  destinationName: string
}

interface OverpassElement {
  type: string
  id: number
  lat?: number
  lon?: number
  tags?: { name?: string }
  center?: { lat: number; lon: number }
}

const PARTNER_COLOR = '#9C6318'  // ochre escuro — contraste sobre tiles claros
const POI_COLOR = '#6B7280'      // gray-500

async function fetchPOIs(
  lat: number,
  lng: number,
  radiusMeters = 2000
): Promise<OverpassElement[]> {
  const query = `
    [out:json][timeout:10];
    (
      node["tourism"="hotel"](around:${radiusMeters},${lat},${lng});
      node["amenity"="restaurant"](around:${radiusMeters},${lat},${lng});
      node["amenity"="bar"](around:${radiusMeters},${lat},${lng});
      node["amenity"="car_rental"](around:${radiusMeters},${lat},${lng});
      way["tourism"="hotel"](around:${radiusMeters},${lat},${lng});
      way["amenity"="restaurant"](around:${radiusMeters},${lat},${lng});
      way["amenity"="bar"](around:${radiusMeters},${lat},${lng});
      way["amenity"="car_rental"](around:${radiusMeters},${lat},${lng});
    );
    out center;
  `
  const controller = new AbortController()
  const timeout = setTimeout(() => controller.abort(), 10_000)

  try {
    const res = await fetch('https://overpass-api.de/api/interpreter', {
      method: 'POST',
      body: query,
      signal: controller.signal,
    })
    clearTimeout(timeout)
    if (!res.ok) return []
    const data = await res.json()
    return (data.elements ?? []) as OverpassElement[]
  } catch {
    clearTimeout(timeout)
    return [] // degradação graciosa — retorna vazio, sem throw
  }
}

function createCircleMarkerElement(color: string, sizePx: number): HTMLDivElement {
  const el = document.createElement('div')
  el.style.cssText = [
    `width: ${sizePx}px`,
    `height: ${sizePx}px`,
    'border-radius: 50%',
    `background: ${color}`,
    `border: ${sizePx <= 10 ? '1px' : '2px'} solid #fff`,
    'box-shadow: 0 1px 4px rgba(0,0,0,0.3)',
    'cursor: pointer',
  ].join(';')
  return el
}

export default function MapWidget({ lat, lng, partners, destinationName }: MapWidgetProps) {
  const containerRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    if (!containerRef.current) return

    let map: import('maplibre-gl').Map | null = null

    async function initMap() {
      // Import dinâmico dentro do useEffect — evita problemas com SSR mesmo em 'use client'
      const maplibregl = (await import('maplibre-gl')).default
      await import('maplibre-gl/dist/maplibre-gl.css')

      if (!containerRef.current) return

      map = new maplibregl.Map({
        container: containerRef.current,
        style: `/api/tiles/maps/streets-v2/style.json`,
        center: [lng, lat], // maplibre: [longitude, latitude]
        zoom: 13,
        attributionControl: false,
      })

      map.addControl(new maplibregl.NavigationControl(), 'bottom-right')
      map.addControl(new maplibregl.AttributionControl({ compact: true }), 'bottom-left')

      map.on('load', async () => {
        if (!map) return
        const maplibre = (await import('maplibre-gl')).default

        // 1. Marcadores de parceiros CAPI (ochre, 14px)
        for (const partner of partners) {
          const el = createCircleMarkerElement(PARTNER_COLOR, 14)

          const popup = new maplibre.Popup({ offset: 10, closeButton: false })
            .setHTML(`<span style="font:600 14px/1.2 var(--font-body,sans-serif);color:var(--stone-900,#1c1917)">${partner.name}</span>`)

          new maplibre.Marker({ element: el })
            .setLngLat([partner.lng, partner.lat])
            .setPopup(popup)
            .addTo(map)
        }

        // 2. POIs do Overpass (cinza, 10px) — degradação graciosa se falhar
        const pois = await fetchPOIs(lat, lng)
        for (const poi of pois) {
          const poiLat = poi.lat ?? poi.center?.lat
          const poiLng = poi.lon ?? poi.center?.lon
          const name = poi.tags?.name
          if (!poiLat || !poiLng || !name) continue

          const el = createCircleMarkerElement(POI_COLOR, 10)

          const popup = new maplibre.Popup({ offset: 8, closeButton: false })
            .setHTML(`<span style="font:600 14px/1.2 var(--font-body,sans-serif);color:var(--stone-900,#1c1917)">${name}</span>`)

          new maplibre.Marker({ element: el })
            .setLngLat([poiLng, poiLat])
            .setPopup(popup)
            .addTo(map)
        }
      })
    }

    initMap().catch((err) => {
      console.error('[MapWidget] init error:', err)
    })

    return () => {
      map?.remove()
    }
  }, [lat, lng, partners, destinationName])

  return (
    <>
      <style>{`
        .map-widget {
          width: 100%;
          height: 400px;
          position: relative;
          overflow: hidden;
          background: var(--stone-900, #1c1917);
        }

        .map-widget__container {
          width: 100%;
          height: 100%;
        }

        /* Popup do MapLibre — sobrescrever estilos padrão */
        .maplibregl-popup-content {
          background: var(--stone-100, #f5f0eb) !important;
          border: 1px solid rgba(0,0,0,0.12) !important;
          border-radius: 4px !important;
          box-shadow: 0 2px 8px rgba(0,0,0,0.15) !important;
          padding: 8px !important;
        }

        .maplibregl-popup-tip {
          display: none !important;
        }

        /* Mobile: ocultar controles de zoom — D-07 */
        @media (max-width: 767px) {
          .maplibregl-ctrl-bottom-right,
          .maplibregl-ctrl-bottom-left {
            display: none !important;
          }
        }
      `}</style>
      <div className="map-widget" aria-label={`Mapa de ${destinationName}`}>
        <div ref={containerRef} className="map-widget__container" />
      </div>
    </>
  )
}
```

NOTAS DE IMPLEMENTAÇÃO:
- `center: [lng, lat]` — MapLibre usa [longitude, latitude], não [lat, lng]
- URL de tiles: `/api/tiles/maps/streets-v2/style.json` — o proxy repassa para `https://api.maptiler.com/maps/streets-v2/style.json?key=...`
- Import CSS do maplibre dentro do useEffect (não no topo do módulo) — necessário para SSR safety mesmo com 'use client'
- `fetchPOIs` retorna array vazio em qualquer erro — não throw, nunca bloqueia o mapa
- Cleanup: `map?.remove()` no return do useEffect evita memory leak
