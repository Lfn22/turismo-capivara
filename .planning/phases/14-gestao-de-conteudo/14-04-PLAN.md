```typescript
{
  preHandler: [authenticate, authorize(['ADMIN'])],
  schema: { body: ApprovalUpdateInput }
}
handler: async (request, reply) => {
  const { id } = request.params
  const { approvalStatus, rejectionReason } = request.body
  
  try {
    const input = ApprovalUpdateInput.parse(request.body)
  } catch (err) {
    return reply.status(400).send({ error: 'Dados inválidos' })
  }
  
  let destination
  if (approvalStatus === 'APPROVED') {
    destination = await destinationService.approveDestination(id)
  } else {
    destination = await destinationService.rejectDestination(id, rejectionReason)
  }
  
  return reply.status(200).send(destination)
}
```

**GET /admin/destinations/pending** — Get list of PENDING destinations awaiting approval
```typescript
{
  preHandler: [authenticate, authorize(['ADMIN'])],
  schema: {
    querystring: {
      limit: z.number().default(50),
      offset: z.number().default(0),
    }
  }
}
handler: async (request, reply) => {
  const { limit, offset } = request.query
  
  const destinations = await prisma.destination.findMany({
    where: { approvalStatus: 'PENDING' },
    include: { createdBy: { select: { id, name, email, tenant: { select: { name, slug } } } } },
    orderBy: { createdAt: 'asc' },
    take: limit,
    skip: offset,
  })
  
  const total = await prisma.destination.count({
    where: { approvalStatus: 'PENDING' }
  })
  
  return reply.status(200).send({
    destinations,
    total,
    limit,
    offset,
  })
}
```

This endpoint supports pagination for Wave 4 super-admin UI.
