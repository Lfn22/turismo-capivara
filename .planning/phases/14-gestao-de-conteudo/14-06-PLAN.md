Create PhotoUploadArea.tsx reusable component:

```typescript
import React, { useCallback, useRef, useState } from 'react'
import { toast } from 'sonner'

interface PhotoUploadAreaProps {
  photos: string[]
  onAdd: (file: File) => Promise<void>
  onRemove: (url: string) => void
  maxPhotos?: number
  uploading?: boolean
}

export function PhotoUploadArea({
  photos,
  onAdd,
  onRemove,
  maxPhotos = 5,
  uploading = false
}: PhotoUploadAreaProps) {
  const fileInputRef = useRef<HTMLInputElement>(null)
  const [isUploading, setIsUploading] = useState(uploading)

  const handleFileSelect = useCallback(async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file) return

    // Validate format
    if (!['image/jpeg', 'image/png', 'image/webp'].includes(file.type)) {
      toast.error('Formato inválido. Use JPEG, PNG ou WebP.')
      return
    }

    // Validate size (5MB)
    if (file.size > 5242880) {
      toast.error('Arquivo muito grande. Máximo 5MB.')
      return
    }

    setIsUploading(true)
    try {
      await onAdd(file)
      toast.success('Foto adicionada com sucesso.')
    } catch (err) {
      toast.error('Falha ao fazer upload. Tente novamente.')
    } finally {
      setIsUploading(false)
    }

    // Reset input
    if (fileInputRef.current) fileInputRef.current.value = ''
  }, [onAdd])

  return (
    <div>
      <div
        style={{
          display: 'grid',
          gridTemplateColumns: 'repeat(auto-fill, minmax(80px, 1fr))',
          gap: '8px',
          marginBottom: '16px'
        }}
      >
        {photos.map((url, idx) => (
          <div
            key={idx}
            style={{
              position: 'relative',
              width: '80px',
              height: '80px',
              borderRadius: '4px',
              overflow: 'hidden',
            }}
          >
            <img
              src={url}
              alt={`Foto ${idx + 1}`}
              style={{
                width: '100%',
                height: '100%',
                objectFit: 'cover'
              }}
            />
            <button
              onClick={() => onRemove(url)}
              style={{
                position: 'absolute',
                top: '4px',
                right: '4px',
                width: '20px',
                height: '20px',
                borderRadius: '50%',
                background: 'rgba(0,0,0,0.6)',
                color: 'white',
                fontSize: '12px',
                border: 'none',
                cursor: 'pointer',
                minHeight: '44px',
                minWidth: '44px',
                display: 'flex',
                alignItems: 'center',
                justifyContent: 'center',
                padding: 0
              }}
              aria-label="Remover foto"
            >
              ×
            </button>
          </div>
        ))}

        {photos.length < maxPhotos && !isUploading && (
          <button
            onClick={() => fileInputRef.current?.click()}
            style={{
              width: '80px',
              height: '80px',
              border: '2px dashed var(--stone-300)',
              background: 'var(--stone-50)',
              borderRadius: '4px',
              cursor: 'pointer',
              display: 'flex',
              flexDirection: 'column',
              alignItems: 'center',
              justifyContent: 'center',
              fontSize: '12px',
              color: 'var(--stone-400)',
              padding: 0,
              minHeight: '80px',
              minWidth: '80px'
            }}
          >
            +
          </button>
        )}

        {isUploading && (
          <div
            style={{
              width: '80px',
              height: '80px',
              borderRadius: '4px',
              background: 'var(--stone-100)',
              display: 'flex',
              alignItems: 'center',
              justifyContent: 'center',
              fontSize: '12px',
              color: 'var(--stone-500)',
              opacity: 0.5
            }}
          >
            Enviando...
          </div>
        )}
      </div>

      {photos.length === maxPhotos && !isUploading && (
        <p style={{ fontSize: '14px', color: 'var(--stone-500)', marginTop: '8px' }}>
          {photos.length}/{maxPhotos} fotos adicionadas
        </p>
      )}

      <input
        ref={fileInputRef}
        type="file"
        accept="image/jpeg,image/png,image/webp"
        style={{ display: 'none' }}
        onChange={handleFileSelect}
        disabled={isUploading}
      />
    </div>
  )
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Form input → API | Zod validation on server catches invalid data (client validation is UX only) |
| User ID → Ownership | currentUser.id compared to destination.createdById (set from JWT) |
| File upload → PhotoUploadArea | Format/size validation in PhotoUploadArea before API call; server re-validates |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-14-24 | Tampering (T) | Client-side form validation bypassed | Mitigate | Server validates all inputs with Zod (schema from Wave 1, Plan 02) |
| T-14-25 | Information Disclosure (I) | User sees destination status badge | Accept | Status badge shows public information (approval status is visible to guide) |
| T-14-26 | Authorization (A) | Non-owner tries to edit/delete via form | Mitigate | Form submit checked; server re-checks ownership before update |
| T-14-27 | Denial of Service (D) | Rapid photo uploads | Accept | No rate limiting in MVP (Wave 2 endpoint has no rate limiting) |

**Disposition rationale:**
- T-14-24/26: Mitigate via Zod + ownership checks (server-side definitive)
- T-14-25: Accept — status is intentionally visible to guide
- T-14-27: Accept — no rate limiting in MVP; can add in v2

</threat_model>
