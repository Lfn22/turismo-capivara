2. **Handle approval action optimistically:**
   ```typescript
   const handleApprove = async (destinationId: string) => {
     // Optimistic: remove from list immediately
     setDestinations(prev => prev.filter(d => d.id !== destinationId))
     
     try {
       const res = await fetch(`/api/destinations/${destinationId}/approve`, {
         method: 'PATCH',
         headers: { 'Content-Type': 'application/json' },
         body: JSON.stringify({ approvalStatus: 'APPROVED' })
       })
       
       if (res.ok) {
         toast.success('Destino aprovado com sucesso.')
       } else {
         // Rollback on error
         const approvalRes = await fetch(`/api/admin/destinations/pending`)
         const { destinations } = await approvalRes.json()
         setDestinations(destinations)
         toast.error('Falha ao aprovar. Tente novamente.')
       }
     } catch (err) {
       toast.error('Erro na conexão. Tente novamente.')
     }
   }
   ```

3. **Similar handler for reject:**

4. **Disable buttons during request:**

   - Show spinner or disable state
